  function buildFloorLabel(
    floorIndex: number,
    job: Job
  ): string {
    
    // First: detect format from existing rooms
    // in this job's signs table
    const sampleFloor = await db.queryOne(\`
      SELECT floor FROM signs
      WHERE job_id = $1
      AND floor IS NOT NULL
      AND floor != ''
      AND floor NOT LIKE 'STAIR%'
      AND floor NOT LIKE 'EXIT%'
      AND floor NOT LIKE 'EVAC%'
      AND floor NOT LIKE 'AOR%'
      AND source != 'rules'
      LIMIT 1
    \`, [job.id]);
    
    if (sampleFloor?.floor) {
      // Extract prefix from existing floor label
      // 'LEVEL 1' → prefix='LEVEL ', num=1
      // 'Floor 2' → prefix='Floor ', num=2
      // 'FL-3'    → prefix='FL-', num=3
      // '2'       → prefix='', num=2
      const match = sampleFloor.floor.match(
        /^([A-Za-z\s\-]*)(\d+)$/
      );
      if (match) {
        const prefix = match[1]; // 'LEVEL ' etc
        return \`\${prefix}\${floorIndex}\`;
      }
    }
    
    // Fallback if no existing rooms yet
    // Use job's detected floor format or default
    return job.floor_label_format
      ? job.floor_label_format.replace('{n}', 
          String(floorIndex))
      : \`LEVEL \${floorIndex}\`;
  }

FIX 3 — floorsServed detecting wrong count
         All building types, all plans
════════════════════════════════════════════════
Fox Hill shows 3 floors (should be 2) because
egress generator rows from previous runs are
being counted as real floors.

Fix detectFloorCount to exclude egress generator
rows from the floor count:

  async function detectFloorCount(
    jobId: string
  ): Promise<number> {
    
    const result = await db.query(\`
      SELECT COUNT(DISTINCT floor) as floor_count
      FROM signs
      WHERE job_id = $1
      AND floor IS NOT NULL
      AND floor != ''
      AND floor ~ '^[A-Za-z\s]*\d+$'
      AND source != 'rules'
      AND room_number NOT LIKE 'STAIR-%'
      AND room_number NOT LIKE 'EXIT-%'
      AND room_number NOT LIKE 'EVAC-%'
      AND room_number NOT LIKE 'AOR-%'
      AND room_number NOT LIKE 'EV-%'
    \`, [jobId]);
    
    const count = parseInt(
      result.rows[0]?.floor_count
    ) || 0;
    
    console.log('[FLOOR-COUNT]', count, 
      'floors detected for job', jobId);
    
    return count > 0 ? count : 2; // min fallback 2
  }

  Residential 6-floor:  returns 6
