CHANGE 2 — Pipeline: process ALL uploaded PDFs
════════════════════════════════════════════════
Currently only processes files from one zone.
Must process ALL uploaded files regardless
of which zone they came from.

In the pipeline file loading step:
  
  // Process ALL uploaded files for this job
  const allFiles = await db.query(\`
    SELECT * FROM job_files
    WHERE job_id = $1
    AND file_type IN ('pdf', 'application/pdf')
    ORDER BY uploaded_at ASC
  \`, [jobId]);

  // Process each file through sheet classifier
  for (const file of allFiles) {
    const sheets = await classifyAllSheets(file);
    // Each sheet gets classified:
    // floor_plan, sign_details, signage_schedule,
    // specialty_signage, egress, skip
    allClassifiedSheets.push(...sheets);
  }

  // Then process by type:
  const floorPlans = allClassifiedSheets
    .filter(s => s.type === 'floor_plan');
  const signDetails = allClassifiedSheets
    .filter(s => s.type === 'sign_details' || 
                 s.type === 'signage_schedule');
  const specialty = allClassifiedSheets
    .filter(s => s.type === 'specialty_signage');

  // Extract rooms from ALL floor plans
  for (const sheet of floorPlans) {
    await extractRooms(sheet, jobId);
  }
  
  // Extract sign schedule from ALL detail sheets
  for (const sheet of signDetails) {
    await extractSignSchedule(sheet, jobId);
  }
  
  // Extract specialty from ALL specialty sheets
  for (const sheet of specialty) {
    await extractSpecialtySigns(sheet, jobId);
  }
