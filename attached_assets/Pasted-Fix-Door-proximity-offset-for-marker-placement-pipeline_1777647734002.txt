What's broken: Room markers are placed at the room label center (middle of the room). For estimators to visually verify sign placement, markers should appear near the door — offset toward the nearest corridor.
Add this function near the coordinate utility functions in pipeline.ts:
typescript/**
 * Offsets a room's marker coordinates toward the nearest corridor.
 * This moves markers from room center to near the door for visual clarity.
 * Falls back to original coordinates if no corridor found nearby.
 */
function applyDoorProximityOffset(
  roomCoordX: number,
  roomCoordY: number,
  corridorRooms: Array<{ coordX: number | null; coordY: number | null }>,
  offsetFraction = 0.20  // move 20% toward nearest corridor
): { x: number; y: number } {
  
  if (corridorRooms.length === 0) {
    return { x: roomCoordX, y: roomCoordY };
  }
  
  // Find nearest corridor by Euclidean distance
  let nearestCorridor: { coordX: number; coordY: number } | null = null;
  let minDistance = Infinity;
  
  for (const corridor of corridorRooms) {
    if (!corridor.coordX || !corridor.coordY) continue;
    const dx = corridor.coordX - roomCoordX;
    const dy = corridor.coordY - roomCoordY;
    const distance = Math.sqrt(dx * dx + dy * dy);
    
    // Only consider corridors within 30% of sheet width (30,000 units)
    // Prevents matching corridors on completely different parts of the plan
    if (distance < minDistance && distance < 30_000) {
      minDistance = distance;
      nearestCorridor = { coordX: corridor.coordX, coordY: corridor.coordY };
    }
  }
  
  if (!nearestCorridor) {
    return { x: roomCoordX, y: roomCoordY };
  }
  
  // Offset toward nearest corridor by offsetFraction
  const offsetX = Math.round(
    roomCoordX + (nearestCorridor.coordX - roomCoordX) * offsetFraction
  );
  const offsetY = Math.round(
    roomCoordY + (nearestCorridor.coordY - roomCoordY) * offsetFraction
  );
  
  // Clamp to valid range
  return {
    x: Math.max(1_000, Math.min(99_000, offsetX)),
    y: Math.max(1_000, Math.min(99_000, offsetY)),
  };
}
