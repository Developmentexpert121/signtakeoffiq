await wp(3.5 as unknown as number, "Merging PDF text + AI vision results");

for (const sheet of relevantSheets) {
  const file = files.find(f => f.id === sheet.fileId);
  if (!file) continue;

  let pdfBuffer: Buffer;
  try {
    pdfBuffer = await downloadFromStorage(file.storagePath);
  } catch { continue; }

  if (!sidecarOk) continue;

  try {
    // Extract words from PDF using sidecar
    const wordResult = await extractWords(pdfBuffer, sheet.pdfPage ?? 1, file.filename);
    if (!wordResult?.words?.length) continue;

    const pageWidth = wordResult.pageWidth ?? 1000;
    const pageHeight = wordResult.pageHeight ?? 1000;

    // Extract rooms from words using existing deterministic logic
    const textRooms = extractRoomsFromWords(wordResult.words, pageWidth, pageHeight);

    logger.log(`[pipeline] Two-pass: sheet ${sheet.sheetId} — ${textRooms.length} rooms from text extraction`);

    for (const textRoom of textRooms) {
      const roomNumLower = textRoom.roomNumber.toLowerCase().trim();
      const roomNameLower = textRoom.roomName.toLowerCase().trim();

      // Find matching vision room
      const visionMatch = extractedRooms.find(r => {
        const vNum = r.roomNumber.toLowerCase().trim();
        const vName = r.roomName.toLowerCase().trim();
        return (roomNumLower && vNum === roomNumLower) ||
               (roomNameLower.length > 3 && vName.includes(roomNameLower)) ||
               (roomNameLower.length > 3 && roomNameLower.includes(vName));
      });

      if (visionMatch) {
        // Both text and vision found this room — upgrade confidence
        visionMatch.aiConfidence = "0.92";
        visionMatch.confirmedByText = true;
        logger.log(`[pipeline] Two-pass: ${textRoom.roomNumber} ${textRoom.roomName} — confirmed by both passes (0.92)`);
      } else {
        // Text found a room vision missed — add it
        extractedRooms.push({
          roomNumber: textRoom.roomNumber,
          roomName: textRoom.roomName,
          level: sheet.level ?? "LEVEL 1",
          x: textRoom.x,
          y: textRoom.y,
          sheetDbId: sheet.id!,
          aiVision: false,
          aiConfidence: "0.85",
          aiIsRestroom: false,
          source: "pdf_text",
        });
        logger.log(`[pipeline] Two-pass: ${textRoom.roomNumber} ${textRoom.roomName} — text only (0.85), added to inventory`);
      }
    }
  } catch (err) {
    logger.warn(`[pipeline] Two-pass text extraction failed for ${sheet.sheetId}: ${err}`);
  }
}

logger.log(`[pipeline] Two-pass complete: ${extractedRooms.length} total rooms after merge`);
