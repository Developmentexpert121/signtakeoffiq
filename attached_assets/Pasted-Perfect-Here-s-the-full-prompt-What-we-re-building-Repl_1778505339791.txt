onDrop={(e) => {
  e.preventDefault()
  const signId = e.dataTransfer.getData('roomId')
  const rect = canvasRef.current.getBoundingClientRect()
  
  // Convert drop position to 0–100,000 grid
  // Account for pan offset and zoom scale
  const dropX = (e.clientX - rect.left - panOffset.x) / displayScale
  const dropY = (e.clientY - rect.top - panOffset.y) / displayScale
  const coordX = Math.round((dropX / imgSize.w) * 100000)
  const coordY = Math.round((dropY / imgSize.h) * 100000)
  
  // Clamp to valid range
  const finalX = Math.max(0, Math.min(100000, coordX))
  const finalY = Math.max(0, Math.min(100000, coordY))
  
  // PATCH sign with new position + sheet_id
  await patchSign(signId, { 
    markerX: finalX, 
    markerY: finalY,
    sheetId: activeSheet.id 
  })
  
  // Invalidate queries
  invalidateListSigns()
  invalidateExport() // triggers XLSX regeneration
}}
