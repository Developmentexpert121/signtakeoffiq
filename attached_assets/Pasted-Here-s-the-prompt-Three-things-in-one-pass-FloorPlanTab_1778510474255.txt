typescriptonDrop={async (e) => {
  e.preventDefault()
  const signId = e.dataTransfer.getData('signId')
  if (!signId || !imgSize || !activeSheet) return

  // Get canvas element bounds in screen space
  const canvasEl = transformContentRef.current
  if (!canvasEl) return
  const rect = canvasEl.getBoundingClientRect()

  // Drop position relative to the image element
  // rect already accounts for pan and zoom via CSS transform
  const relX = e.clientX - rect.left
  const relY = e.clientY - rect.top

  // Convert to 0–100,000 grid
  // rect.width = imgSize.w * displayScale, so:
  const coordX = Math.round((relX / rect.width) * 100000)
  const coordY = Math.round((relY / rect.height) * 100000)

  const finalX = Math.max(0, Math.min(100000, coordX))
  const finalY = Math.max(0, Math.min(100000, coordY))

    markerX: finalX,
    markerY: finalY,
    sheetId: activeSheet.id
  })

  invalidateListSigns()
}}

typescript// If imgSize not known yet, return all non-sentinel signs
if (!imgSize) {

// Expanded buffer — 15% on each side
const vLeft   = (-panOffset.x - containerW * 0.15) / displayScale
const vRight  = (-panOffset.x + containerW * 1.15) / displayScale
const vTop    = (-panOffset.y - containerH * 0.15) / displayScale
const vBottom = (-panOffset.y + containerH * 1.15) / displayScale

// Exclude null and sentinel coordinates from canvas render
// (they live in the tray, not on the plan)
return sheetSigns.filter(s => {
  if (s.isDeleted) return false
  if (s.markerX === null || s.markerX === 50000) return false
  const pixX = (s.markerX / 100000) * imgSize.w
  const pixY = (s.markerY / 100000) * imgSize.h
  return pixX >= vLeft && pixX <= vRight &&
         pixY >= vTop  && pixY <= vBottom
}).slice(0, _MAX_MARKERS)
