// shared/room-validation.ts

const VALID_ROOM_NUMBER_PATTERNS: Record<string, RegExp[]> = {
  education: [
    /^[WE]C?\d{3}[A-Z]?$/,   // W123, E456A, WC100, EC110A
    /^S[ABCD]\d{2}$/,          // SA01, SB02
    /^EV\d{2}$/,               // EV01
  ],
  healthcare:   [/^\d{3,4}[A-Z]?$/, /^[A-Z]{1,3}\d{3}$/],
  office:       [/^\d{3,4}[A-Z]?$/, /^[A-Z]\d{3}$/],
  government:   [/^\d{3,4}[A-Z]?$/, /^[A-Z]-?\d{3}$/],
  hotel:        [/^\d{3,4}[A-Z]?$/],
  residential:  [/^\d{3,4}[A-Z]?$/],
  assembly:     [/^\d{3,4}[A-Z]?$/, /^[A-Z]{1,2}-?\d{2,3}$/],
  unknown:      [/^.{2,10}$/],  // permissive fallback
};

const GARBLED_NAME_PATTERNS = [
  /\d+X\d+/,             // 60X21, 48X21 (casework dims)
  /\d+'-\d/,             // 8'-0'' (imperial dims)
  /TO [EW]\d/,           // E119C.1 TO E119C.2
  /^[.'"\-\s\d]+$/,      // only punctuation/numbers
  /(\.\d){2,}/,          // .1 .1 .1 (callout refs)
  /\b19[A-J]\b/,         // 19A 19B (LEED sign codes)
  /\bMB\b|\bTB\b|\bCB\b/, // marker/tackboard callouts
];

export function isValidRoomNumber(
  num: string, 
  buildingType: string
): boolean {
  const patterns = VALID_ROOM_NUMBER_PATTERNS[buildingType] 
    ?? VALID_ROOM_NUMBER_PATTERNS.unknown;
  return patterns.some(p => p.test(num));
}

export function isValidRoomName(name: string): boolean {
  if (!name || name.trim().length < 2) return false;
  if (GARBLED_NAME_PATTERNS.some(p => p.test(name))) return false;
  // Must be at least 40% alpha characters
  const alphaCount = (name.match(/[a-zA-Z]/g) || []).length;
  return alphaCount / name.length >= 0.4;
}
