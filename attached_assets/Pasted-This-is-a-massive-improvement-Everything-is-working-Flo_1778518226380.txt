typescriptconst panToMarker = (sign: Sign) => {

  const pixX = (sign.markerX / 100000) * imgSize.w
  const pixY = (sign.markerY / 100000) * imgSize.h
  // Use react-zoom-pan-pinch's setTransform to center on the marker

    -pixY * 1.5 + containerH / 2,

  )
  setHighlightedSignId(sign.id)
