Context: Large PDFs (15MB+) timeout during rasterization at 250 DPI. The fix is to rasterize at 150 DPI (faster, reliable) then tile the resulting PNG into 4 sections before sending to Gemini. Each tile is 2× zoomed relative to the full image, giving Gemini readable text at 1/32" scale without the rasterization timeout.
Fix 1 — Add tile splitting to sidecar (artifacts/sidecar/main.py)
Add a new /rasterize-tiles endpoint that:

Rasterizes the requested page at 150 DPI
Splits the PNG into a 2×2 grid (4 tiles)
Returns all 4 tiles as base64 with their offsets

python@app.route('/rasterize-tiles', methods=['POST'])
def rasterize_tiles():
    data = request.json
    pdf_bytes = base64.b64decode(data['pdf_base64'])
    page_num = data.get('page', 1) - 1  # 0-indexed
    dpi = data.get('dpi', 150)
    
    doc = fitz.open(stream=pdf_bytes, filetype='pdf')
    page = doc[page_num]
    mat = fitz.Matrix(dpi/72, dpi/72)
    pix = page.get_pixmap(matrix=mat)
    
    img_bytes = pix.tobytes('png')
    from PIL import Image
    import io
    img = Image.open(io.BytesIO(img_bytes))
    w, h = img.size
    
    # 2×2 grid with 10% overlap
    overlap_x = int(w * 0.05)
    overlap_y = int(h * 0.05)
    
    tiles = []
    for row in range(2):
        for col in range(2):
            x1 = max(0, col * w//2 - overlap_x)
            y1 = max(0, row * h//2 - overlap_y)
            x2 = min(w, (col+1) * w//2 + overlap_x)
            y2 = min(h, (row+1) * h//2 + overlap_y)
            
            tile = img.crop((x1, y1, x2, y2))
            buf = io.BytesIO()
            tile.save(buf, format='PNG')
            
            tiles.append({
                'base64': base64.b64encode(buf.getvalue()).decode(),
                'offset_x': x1 / w,
                'offset_y': y1 / h,
                'scale_x': (x2 - x1) / w,
                'scale_y': (y2 - y1) / h,
                'col': col,
                'row': row,
            })
    
    return jsonify({'tiles': tiles, 'width': w, 'height': h})
Fix 2 — Add rasterizeTiles function to sidecar-client.ts
typescriptexport interface RasterTile {
  base64: string;
  offsetX: number;
  offsetY: number;
  scaleX: number;
  scaleY: number;
  col: number;
  row: number;
}

export async function rasterizeTiles(
  pdfBuf: Buffer,
  pageNumber: number,
  dpi = 150
): Promise<RasterTile[]> {
  const response = await callSidecarWithRetry(
    `${SIDECAR_URL}/rasterize-tiles`,
    {
      pdf_base64: pdfBuf.toString('base64'),
      page: pageNumber,
      dpi,
    }
  );
  const data = await response.json() as { tiles: Array<{
    base64: string;
    offset_x: number;
    offset_y: number;
    scale_x: number;
    scale_y: number;
    col: number;
    row: number;
  }>};
  return data.tiles.map(t => ({
    base64: t.base64,
    offsetX: t.offset_x,
    offsetY: t.offset_y,
    scaleX: t.scale_x,
    scaleY: t.scale_y,
    col: t.col,
    row: t.row,
  }));
}
