**Part 2 — Example retrieval function in `training.ts`:**

Add a new exported function `getTrainingContext(tenantId, buildingType)` that:

1. Fetches all `approved` patterns for the tenant from `training_patterns`
2. Fetches the top 3 most similar validated import snapshots from `import_snapshots` by matching `building_type` and highest `accuracy_score`
3. Returns a formatted string for injection into the Claude prompt:

```ts
export async function getTrainingContext(
  tenantId: string,
  buildingType: string | null,
): Promise<string> {
  // Get approved patterns
  const patterns = await db.select()
    .from(trainingPatternsTable)
    .where(and(
      eq(trainingPatternsTable.tenantId, tenantId),
      eq(trainingPatternsTable.status, "approved"),
    ))
    .orderBy(desc(trainingPatternsTable.evidenceCount))
    .limit(15);

  // Get similar validated examples
  const examples = await db.select()
    .from(importSnapshotsTable)
    .where(and(
      eq(importSnapshotsTable.tenantId, tenantId),
      isNotNull(importSnapshotsTable.accuracyScore),
      ...(buildingType ? [eq(importSnapshotsTable.buildingType, buildingType)] : []),
    ))
    .orderBy(desc(importSnapshotsTable.accuracyScore))
    .limit(3);

  if (patterns.length === 0 && examples.length === 0) return "";

  let context = "TRAINING CONTEXT — patterns validated by human estimators:\n\n";

  if (patterns.length > 0) {
    context += "KNOWN SIGN TYPE RULES (apply these during extraction):\n";
    for (const p of patterns) {
      context += `- ${p.description} → ${p.suggestedFix}\n`;
    }
    context += "\n";
  }

  if (examples.length > 0) {
    context += "VALIDATED REFERENCE JOBS (similar building type, human-verified):\n";
    for (const e of examples) {
      const score = e.accuracyScore ? `${Math.round(parseFloat(e.accuracyScore) * 100)}%` : "unknown";
      context += `- ${e.buildingType ?? "Unknown type"} project: ${e.totalHumanSigns ?? "?"} signs, accuracy ${score}\n`;
    }
  }

  return context;
}
```
