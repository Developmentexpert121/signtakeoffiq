router.post("/jobs/:jobId/files/drive", requireAuth, async (req, res) => {
  const { tenantId } = req.auth_ctx!;
  const jobId = String(req.params.jobId);
  const { driveUrl, fileCategory } = req.body;

  if (!driveUrl || typeof driveUrl !== "string") {
    res.status(400).json({ error: "driveUrl is required" });
    return;
  }

  // Verify job belongs to tenant
  const [job] = await db.select()
    .from(jobsTable)
    .where(and(eq(jobsTable.id, jobId), eq(jobsTable.tenantId, tenantId)));
  
  if (!job) {
    res.status(404).json({ error: "Job not found" });
    return;
  }

  try {
    const { buffer, filename, contentType, sizeBytes } = 
      await downloadFromGoogleDrive(driveUrl);

    // Upload to GCS
    const path = `uploads/${tenantId}/${jobId}/${Date.now()}-${filename}`;
    const storagePath = await uploadToStorage(path, buffer, contentType, tenantId);

    // Insert into job_files table
    const fileId = newId("file");
    await db.insert(jobFilesTable).values({
      id: fileId,
      jobId,
      tenantId,
      filename,
      storagePath,
      fileCategory: fileCategory ?? "floor_plan",
      contentType,
      sizeBytes,
    });

    res.json({
      id: fileId,
      filename,
      storagePath,
      sizeBytes,
      sizeMB: (sizeBytes / 1024 / 1024).toFixed(1),
      contentType,
    });
  } catch (err) {
    const message = err instanceof Error ? err.message : "Download failed";
    res.status(400).json({ error: message });
  }
});
