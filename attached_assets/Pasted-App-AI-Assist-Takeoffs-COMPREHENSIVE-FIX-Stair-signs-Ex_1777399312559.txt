PART B — Core fix: Egress Sign Generator
         New module — runs AFTER room extraction
════════════════════════════════════════════════
Create: /api-server/src/lib/egress-sign-generator.ts

This module runs after room extraction completes
and generates stair, exit, evac map, area of
rescue, and max occupancy signs from metadata.
It does NOT depend on room extraction finding
stair or exit rooms.

async function generateEgressSigns(job, profile) {

  // ── STEP 1: Detect stair count ──────────────
  // Try extracted rooms first, fall back to default
  
  const extractedStairs = await db.query(
    \`SELECT DISTINCT room_number, room_name, floor
     FROM signs WHERE job_id = $1
     AND (
       room_name ILIKE ANY($2)
       OR room_number ~ '^S[A-F]\\d{2}$'
       OR room_number ~ '^S\\d{1,2}$'
     )\`,
    [job.id, profile.stair_name_patterns]
  );

  const stairCount = extractedStairs.length > 0
    ? new Set(extractedStairs.map(s => 
        s.room_number.replace(/\\d+$/, '')
      )).size
    : profile.stair_count_default;

  const floorsServed = job.floor_count || 2;
  const isEstimated = extractedStairs.length === 0;

  // ── STEP 2: Generate stair signs ────────────
  // R11: 1 corridor + 1 landing per stair per floor
  
  for (let s = 1; s <= stairCount; s++) {
    const stairLabel = stairCount <= 4
      ? String.fromCharCode(64 + s)  // A, B, C, D
      : String(s);                    // 1, 2, 3...

    for (let f = 1; f <= floorsServed; f++) {
      await insertEgressSign(job.id, {
        room_number: \`STAIR-\${stairLabel}\`,
        room_name: \`STAIR \${stairLabel}\`,
        floor: \`LEVEL \${f}\`,
        sign_type: 'Stair (Corridor)',
        qty: 1,
        ada_required: true,
        rule_ref: 'R11',
        source: 'rules',
        confidence: isEstimated ? 0.7 : 1.0,
        notes: isEstimated
          ? \`Estimated — verify stair count on plans\`
          : \`Detected from plan extraction\`
      });
      await insertEgressSign(job.id, {
        room_number: \`STAIR-\${stairLabel}\`,
        room_name: \`STAIR \${stairLabel}\`,
        floor: \`LEVEL \${f}\`,
        sign_type: 'Stair (Landing)',
        qty: 1,
        ada_required: true,
        rule_ref: 'R11',
        source: 'rules',
        confidence: isEstimated ? 0.7 : 1.0,
        notes: isEstimated
          ? \`Estimated — verify stair count on plans\`
          : \`Detected from plan extraction\`
      });
    }

    // Area of Rescue — Government and Assembly only
    if (profile.requires_area_of_rescue) {
      for (let f = 1; f <= floorsServed; f++) {
        await insertEgressSign(job.id, {
          room_number: \`STAIR-\${stairLabel}\`,
          room_name: \`STAIR \${stairLabel}\`,
          floor: \`LEVEL \${f}\`,
          sign_type: 'Area of Rescue',
          qty: 2,  // identification + instructions
          ada_required: true,
          rule_ref: 'R16',
          source: 'rules',
          notes: 'Area of Rescue ID + Instruction signs'
        });
      }
    }
  }

  // ── STEP 3: Generate exit signs ─────────────
  // R9: stair doors + exterior doors
  
  // Exit at every stair door on every floor
  for (let s = 1; s <= stairCount; s++) {
    const stairLabel = stairCount <= 4
      ? String.fromCharCode(64 + s) : String(s);
    for (let f = 1; f <= floorsServed; f++) {
      await insertEgressSign(job.id, {
        room_number: \`EXIT-STAIR-\${stairLabel}-L\${f}\`,
        room_name: \`Exit at Stair \${stairLabel}\`,
        floor: \`LEVEL \${f}\`,
        sign_type: 'Exit',
        qty: 1,
        ada_required: false,
        rule_ref: 'R9',
        source: 'rules',
        notes: 'Illuminated exit sign at stair entry door'
      });
    }
  }

  // Exterior exit doors (ground floor only)
  // Use building type minimum
  const minExteriorExits = {
    education: 4,    // 2 per wing × 2 wings
    healthcare: 4,
    commercial: 2,
    government: 3,
    hotel: 2,
    residential: 2,
    assembly: 4,     // min 3 per IBC Group A
    unknown: 2
  }[job.building_type] || 2;

  for (let e = 1; e <= minExteriorExits; e++) {
    await insertEgressSign(job.id, {
      room_number: \`EXIT-EXT-\${e}\`,
      room_name: \`Exterior Exit Door \${e}\`,
      floor: 'LEVEL 1',
      sign_type: 'Exit',
      qty: 1,
      ada_required: false,
      rule_ref: 'R9',
      source: 'rules',
      notes: 'Verify count — exterior exit doors on plans'
    });
  }

  // Remove Exit signs from extracted rooms
  // (MECHANICAL, LOBBY etc should not have Exit)
  await db.query(
    \`DELETE FROM signs
     WHERE job_id = $1
     AND sign_type = 'Exit'
     AND source != 'rules'\`,
    [job.id]
  );

  // ── STEP 4: Generate evacuation maps ────────
  // R13: decision points only, max per floor cap
  
  // Remove all existing evac maps first
  await db.query(
    \`DELETE FROM signs
     WHERE job_id = $1
     AND sign_type = 'Evacuation Map'\`,
    [job.id]
  );

  // Re-assign to correct rooms only
  const evacCandidates = await db.query(
    \`SELECT id, room_name, floor
     FROM signs
     WHERE job_id = $1
     AND UPPER(room_name) = ANY($2)
     ORDER BY floor, room_name\`,
    [job.id, ['CORRIDOR','MAIN CORRIDOR','LOBBY',
               'MAIN LOBBY','GYM LOBBY','MUSIC LOBBY',
               'ELEVATOR LOBBY','CAFETERIA','GYMNASIUM']]
  );

  const evacByFloor = {};
  for (const candidate of evacCandidates) {
    const floor = candidate.floor || 'LEVEL 1';
    if (!evacByFloor[floor]) evacByFloor[floor] = 0;
    if (evacByFloor[floor] < profile.evac_map_max_per_floor) {
      await insertEgressSign(job.id, {
        room_number: candidate.room_number,
        room_name: candidate.room_name,
        floor: floor,
        sign_type: 'Evacuation Map',
        qty: 1,
        ada_required: false,
        rule_ref: 'R13',
        source: 'rules'
      });
      evacByFloor[floor]++;
    }
  }

  // Fallback: if no corridor/lobby found,
  // place 1 evac map per floor at first room
  for (let f = 1; f <= floorsServed; f++) {
    const floorKey = \`LEVEL \${f}\`;
    if (!evacByFloor[floorKey]) {
      await insertEgressSign(job.id, {
        room_number: \`EVAC-L\${f}\`,
        room_name: \`Main Corridor\`,
        floor: floorKey,
        sign_type: 'Evacuation Map',
        qty: 1,
        ada_required: false,
        rule_ref: 'R13',
        source: 'rules',
        notes: 'Location estimated — verify on plans'
      });
    }
  }

  // ── STEP 5: Max Occupancy for assembly rooms ─
  // R10: assembly rooms get Room ID w/Insert AND
  //      Max Occupancy as separate sign records
  
  const assemblyRooms = await db.query(
    \`SELECT id, room_number, room_name, floor
     FROM signs
     WHERE job_id = $1
     AND is_assembly = true
     AND sign_type != 'Max Occupancy'\`,
    [job.id]
  );

  for (const room of assemblyRooms) {
    // Check if Max Occupancy already exists for this room
    const existing = await db.query(
      \`SELECT id FROM signs
       WHERE job_id = $1
       AND room_number = $2
       AND sign_type = 'Max Occupancy'\`,
      [job.id, room.room_number]
    );
    if (existing.length === 0) {
      await insertEgressSign(job.id, {
        room_number: room.room_number,
        room_name: room.room_name,
        floor: room.floor,
        sign_type: 'Max Occupancy',
        qty: 1,
        ada_required: false,
        rule_ref: 'R10',
        source: 'rules',
        notes: 'Post inside room at primary entrance'
      });
    }
  }
}

// Call this at the end of the job processing pipeline
// after all room extraction and semantic mapping:
await generateEgressSigns(job, buildingTypeProfile);
