Step 3 — Create materialSpec.ts
Create artifacts/api-server/src/lib/materialSpec.ts:
typescriptexport interface JobMaterialSpec {
  jobId:          string;
  substrate:      string | null;
  finishMethod:   string | null;
  brailleSpec:    string | null;
  mountingHeight: string | null;
  manufacturer:   string | null;
  source:         "sign_schedule" | "manual";
}

// In-memory store per job — persists for the lifetime of the server process
// Cleared on force-fresh rescan
const store = new Map<string, JobMaterialSpec>();

export function setJobMaterialSpec(spec: JobMaterialSpec): void {
  store.set(spec.jobId, spec);
}

export function getJobMaterialSpec(jobId: string): JobMaterialSpec | null {
  return store.get(jobId) ?? null;
}

export function clearJobMaterialSpec(jobId: string): void {
  store.delete(jobId);
}
Import and call setJobMaterialSpec(jobMaterialSpec) at the end of the pipeline run if jobMaterialSpec is not null. Call clearJobMaterialSpec(jobId) at the force-fresh clear step alongside the existing sign_schedule_entries delete.
