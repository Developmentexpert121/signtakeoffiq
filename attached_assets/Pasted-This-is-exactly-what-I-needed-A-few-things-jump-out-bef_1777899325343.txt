### Fix 2 — Floor replication from range labels (pipeline.ts)

Find `parseLevelFromLabel()` (or wherever `floorLabel` text on `job_files` is parsed into a level number). Replace or extend it to detect range patterns and return a levels array plus a replicate flag:

```typescript
function parseLevelFromLabel(label: string): { levels: number[]; replicate: boolean } {
  if (!label) return { levels: [], replicate: false };

  // Range patterns: "Floors 3-6", "Levels 3-6", "3rd-6th Floor", "Floor 3-6"
  const rangeMatch = label.match(/(\d+)\s*[-–to]+\s*(\d+)/i);
  if (rangeMatch) {
    const start = parseInt(rangeMatch[1], 10);
    const end   = parseInt(rangeMatch[2], 10);
    if (!isNaN(start) && !isNaN(end) && end > start && (end - start) <= 20) {
      const levels = Array.from({ length: end - start + 1 }, (_, i) => start + i);
      return { levels, replicate: true };
    }
  }

  // Single level: "Floor 3", "Level 2", "3rd Floor"
  const singleMatch = label.match(/(\d+)/);
  if (singleMatch) return { levels: [parseInt(singleMatch[1], 10)], replicate: false };

  return { levels: [], replicate: false };
}
```

if (geminiEntries.length === 0) {
