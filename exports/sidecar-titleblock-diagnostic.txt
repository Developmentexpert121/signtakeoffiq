def _title_block_words(pg):
    """
    Return words in the bottom-right quadrant:
      x0  >= 70% of page width   (right 30%)
      top >= 75% of page height  (bottom 25%)
    This is where architectural title blocks live.
    """
    pw = pg.width  or 1.0
    ph = pg.height or 1.0
    x_thresh = pw * 0.70
    y_thresh = ph * 0.75
    words = pg.extract_words(x_tolerance=3, y_tolerance=3) or []
    return [w for w in words
            if w.get("x0", 0) >= x_thresh
            and w.get("top", w.get("y0", 0)) >= y_thresh]

def _find_sheet_in_title_block(pg):
    pw, ph = pg.width or 1.0, pg.height or 1.0

    # Step 1: find sheet ID in bottom-right corner (right 30%, bottom 25%)
    br_words = _title_block_words(pg)
    if not br_words:
        return None, None

    br_sorted = sorted(br_words, key=lambda w: (top, x0))
    br_text = " ".join(w["text"] for w in br_sorted)   # all words joined into one string

  For each page:
    PRIMARY:  _find_sheet_in_title_block()
      Region: x >= 70% width AND y >= 75% height (bottom-right 30% × 25%)
      Finds:  FIRST AIA-style SHEET_ID_PATTERN match in that region
              (sorted top→bottom, left→right, joined, first match wins)
      Then:   expands to x >= 50%, y >= 50% to find the drawing title

No priority is given to the very bottom-right corner specifically — ALL
words in the bottom 25% / right 30% are pooled together, sorted, and the
FIRST SHEET_ID_PATTERN match in the pooled text is used.

area of the finish schedule (e.g., "C-1 STAFF RESTROOM") match SHEET_ID_PATTERN
and appear before "A-710" in the sorted word list.
